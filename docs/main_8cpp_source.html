79bool validarPrazo(const std::string& prazo) {
80 if (prazo.empty()) return true;
81 std::regex fmt(R"(\d{2}/\d{2}/\d{4})");
82 if (!std::regex_match(prazo, fmt)) return false;
83 int dia = std::stoi(prazo.substr(0, 2));
84 int mes = std::stoi(prazo.substr(3, 2));
85 int ano = std::stoi(prazo.substr(6, 4));
86 if (mes < 1 || mes > 12) return false;
87 if (ano < 1900 || ano > 2100) return false;
88
89 int diasNoMes[] = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
90 bool bissexto = (ano % 4 == 0 && ano % 100 != 0) || (ano % 400 == 0);
91 if (mes == 2 && bissexto) diasNoMes[1] = 29;
92
93 if (dia < 1 || dia > diasNoMes[mes - 1]) return false;
94 return true;
95}
